Which models of aircraft have the longest average flight distance?

flights %>%
  select( distance, tailnum ) %>%
  group_by( tailnum) %>%
  summarize( dist = mean( distance, na.rm=TRUE ) ) %>%
  left_join( planes, by="tailnum" ) %>%
  select( model, dist ) %>%
  arrange( -dist ) %>%
  group_by( model ) %>%
  summarize( dist = mean( dist, na.rm=TRUE ) ) %>%
  arrange( -dist )
